def generate_pe
  pe_orig = Metasm::PE.decode_file(template)
  pe = copy_pe(pe_orig)

  append_section(pe)

  pe.optheader.entrypoint = 'new_entrypoint'
  pe.cpu = pe_orig.cpu

  pe.encode_string
end
